        // Fallback Local NLP Engine (Rule-based & Regex Parsing)
        function analyzeLocally(text, name, studentId, dept) {
            const textLower = text.toLowerCase();
            let category = "Others";
            let recipientEmail = "campus-help@campus.edu";
            let urgency = "Low";
            let slaHours = "72 Hours";
            let sentiment = "Neutral";

            // Category & Authority Email Detection
            if (textLower.includes("hostel") || textLower.includes("ac") || textLower.includes("room") || textLower.includes("water cooler") || textLower.includes("fan")) {
                category = "Hostel & Facilities";
                recipientEmail = "warden@campus.edu";
            } else if (textLower.includes("wifi") || textLower.includes("internet") || textLower.includes("pc") || textLower.includes("network") || textLower.includes("computer")) {
                category = "Wi-Fi & IT Infrastructure";
                recipientEmail = "it-admin@campus.edu";
            } else if (textLower.includes("mess") || textLower.includes("food") || textLower.includes("dining") || textLower.includes("lunch")) {
                category = "Mess & Dining";
                recipientEmail = "mess-committee@campus.edu";
            } else if (textLower.includes("lab") || textLower.includes("gpu") || textLower.includes("graphics") || textLower.includes("equipment")) {
                category = "Academic Labs & Equipment";
                recipientEmail = "lab-incharge@campus.edu";
            }

            // Urgency & SLA Detection
            if (textLower.includes("broken") || textLower.includes("leak") || textLower.includes("down") || textLower.includes("urgent") || textLower.includes("failure") || textLower.includes("stopped")) {
                urgency = "Critical";
                slaHours = "2 Hours";
                sentiment = "Frustrated / Urgent";
            } else if (textLower.includes("slow") || textLower.includes("quality") || textLower.includes("noise")) {
                urgency = "Medium";
                slaHours = "24 Hours";
                sentiment = "Dissatisfied";
            }

            // Entity/Location Extraction using Regex Patterns
            let location = "Campus Premises";
            const roomMatch = text.match(/(Hostel\s*\d+|Block\s*\d+|CS Lab\s*\d+|Lab\s*\d+|Room\s*\d+|PC\s*#?\d+|Dining Hall\s*[A-Z])/i);
            if (roomMatch) {
                location = roomMatch[0];
            }

            const refId = 'GRV-' + Math.floor(1000 + Math.random() * 9000);

            // Draft Formal Escalation Email
            const subject = `ESCALATION [${urgency.toUpperCase()}] [${refId}]: ${category} Issue - ${location}`;
            const body = `OFFICIAL CAMPUS GRIEVANCE ESCALATION NOTICE
--------------------------------------------------
Reference ID: ${refId}
Timestamp: ${new Date().toLocaleString()}

RECIPIENT AUTHORITY: ${recipientEmail}
SUBMITTED BY: ${name} (ID: ${studentId}, Dept: ${dept})

--------------------------------------------------
GRIEVANCE ANALYSIS
--------------------------------------------------
- Category: ${category}
- Urgency Level: ${urgency} (Action Required within SLA: ${slaHours})
- Extracted Node/Location: ${location}
- Detected Sentiment: ${sentiment}

FULL COMPLAINT DESCRIPTION:
"${text}"

--------------------------------------------------
ACTION REQUESTED:
This is an automated priority dispatch generated by the UniSolve AI System. Please assign a campus technician immediately and log the status update in the portal.`;

            return {
                refId,
                category,
                urgency,
                slaHours,
                location,
                sentiment,
                recipientEmail,
                emailDraft: {
                    to: recipientEmail,
                    subject,
                    body
                }
            };
        }

        // Live Gemini API Call or Local Fallback Dispatch
        async function analyzeComplaint(text, name, studentId, dept) {
            if (!state.apiKey) {
                // Return local rule-based analysis
                return analyzeLocally(text, name, studentId, dept);
            }

            // Real Gemini 3 Flash API Payload
            try {
                const systemPrompt = `You are a Smart Campus AI Grievance Classification Engine. Analyze the student complaint and return a valid JSON object with the following fields:
                - category: string ("Hostel & Facilities", "Mess & Dining", "Wi-Fi & IT Infrastructure", "Academic Labs & Equipment", "Others")
                - urgency: string ("Critical", "Medium", "Low")
                - slaHours: string ("2 Hours", "24 Hours", "72 Hours")
                - location: string (extracted room/hostel/lab/node/PC number)
                - sentiment: string ("Urgent/Frustrated", "Neutral", "Mild Concern")
                - recipientEmail: string (appropriate authority email like warden@campus.edu, it-admin@campus.edu, mess-committee@campus.edu)
                - formalSubject: string (an official email subject line)
                - formalBody: string (a formal, professional escalation email body)`;

                const userPrompt = `Student Name: ${name}, ID: ${studentId}, Dept: ${dept}. Complaint Description: "${text}"`;

                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${state.apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: userPrompt }] }],
                        systemInstruction: { parts: [{ text: systemPrompt }] },
                        generationConfig: { responseMimeType: "application/json" }
                    })
                });

                const data = await response.json();
                const resultText = data?.candidates?.[0]?.content?.parts?.[0]?.text;
                if (resultText) {
                    const parsed = JSON.parse(resultText);
                    const refId = 'GRV-' + Math.floor(1000 + Math.random() * 9000);
                    return {
                        refId,
                        category: parsed.category || "Others",
                        urgency: parsed.urgency || "Medium",
                        slaHours: parsed.slaHours || "24 Hours",
                        location: parsed.location || "Campus",
                        sentiment: parsed.sentiment || "Neutral",
                        recipientEmail: parsed.recipientEmail || "admin@campus.edu",
                        emailDraft: {
                            to: parsed.recipientEmail || "admin@campus.edu",
                            subject: parsed.formalSubject || `Grievance Escalation: ${refId}`,
                            body: parsed.formalBody || text
                        }
                    };
                }
            } catch (err) {
                console.warn("Gemini API call failed, falling back to local NLP:", err);
            }

            return analyzeLocally(text, name, studentId, dept);
        }

                timestamp: new Date().toLocaleString(),
                emailDraft: analysis.emailDraft
            };

            // Save Ticket to State
            state.tickets.unshift(newTicket);
            state.currentActiveTicket = newTicket;

            // Update UI
            document.getElementById('aiLoadingState').classList.add('hidden');
            document.getElementById('aiPreviewResult').classList.remove('hidden');

            // Fill Diagnostic Card
            document.getElementById('ticketRefBadge').innerText = newTicket.refId;
            document.getElementById('resCategory').innerHTML = `<i data-lucide="tag" class="w-3.5 h-3.5 text-brand-500"></i> ${newTicket.category}`;
            
            const urgencyColor = newTicket.urgency === 'Critical' ? 'text-red-600' : (newTicket.urgency === 'Medium' ? 'text-amber-500' : 'text-emerald-600');
            document.getElementById('resUrgency').className = `font-bold ${urgencyColor} flex items-center gap-1 mt-0.5`;
            document.getElementById('resUrgency').innerHTML = `<i data-lucide="alert-triangle" class="w-3.5 h-3.5"></i> ${newTicket.urgency} (SLA: ${newTicket.slaHours})`;

            document.getElementById('resLocation').innerHTML = `<i data-lucide="map-pin" class="w-3.5 h-3.5 text-amber-500"></i> ${newTicket.location}`;
            document.getElementById('resSentiment').innerHTML = `<i data-lucide="frown" class="w-3.5 h-3.5 text-orange-500"></i> ${newTicket.sentiment}`;

            document.getElementById('emailSnippetPreview').innerText = `To: ${newTicket.emailDraft.to} | Subject: ${newTicket.emailDraft.subject}`;

            lucide.createIcons();
            renderAdminTable();
            renderTrackedTickets();
            showToast(`Grievance ${newTicket.refId} submitted & analyzed!`);
        }

        // Render Student Tracking Screen
        function renderTrackedTickets(filterQuery = '') {
            const container = document.getElementById('trackedTicketsContainer');
            container.innerHTML = '';

            let list = state.tickets;
            if (filterQuery) {
                const q = filterQuery.toLowerCase();
                list = list.filter(t => t.refId.toLowerCase().includes(q) || t.studentId.toLowerCase().includes(q));
            }

            if (list.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-8 text-slate-400 text-xs">
                        No grievance record found matching your query.
                    </div>`;
                return;
            }

            list.forEach(ticket => {
                const statusBadge = ticket.status === 'Resolved' 
                    ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300'
                    : (ticket.status === 'In Progress' 
                        ? 'bg-blue-100 text-blue-700 dark:bg-blue-900/40 dark:text-blue-300' 
                        : 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300');

                const item = document.createElement('div');
                item.className = 'bg-white dark:bg-slate-800 p-5 rounded-2xl border border-slate-200 dark:border-slate-700 shadow-xs space-y-3';
                item.innerHTML = `
                    <div class="flex flex-wrap justify-between items-center gap-2">
                        <div class="flex items-center space-x-2">
                            <span class="font-mono font-bold text-brand-600 text-sm">${ticket.refId}</span>
                            <span class="text-xs px-2.5 py-0.5 rounded-full font-semibold ${statusBadge}">${ticket.status}</span>
                        </div>
                        <span class="text-[11px] text-slate-400"><i data-lucide="clock" class="w-3 h-3 inline mr-1"></i>${ticket.timestamp}</span>
                    </div>

                    <p class="text-xs text-slate-700 dark:text-slate-300 font-medium">"${ticket.description}"</p>

                    <div class="grid grid-cols-2 sm:grid-cols-4 gap-2 text-[11px] bg-slate-50 dark:bg-slate-900/60 p-2.5 rounded-xl text-slate-500">
                        <div><span class="block text-[10px] text-slate-400">CATEGORY</span><strong class="text-slate-700 dark:text-slate-200">${ticket.category}</strong></div>
                        <div><span class="block text-[10px] text-slate-400">LOCATION</span><strong class="text-slate-700 dark:text-slate-200">${ticket.location}</strong></div>
                        <div><span class="block text-[10px] text-slate-400">TECHNICIAN</span><strong class="text-slate-700 dark:text-slate-200">${ticket.assignedTech}</strong></div>
                        <div><span class="block text-[10px] text-slate-400">TARGET SLA</span><strong class="text-slate-700 dark:text-slate-200">${ticket.slaHours}</strong></div>
                    </div>
                `;
                container.appendChild(item);
            });
            lucide.createIcons();
        }

        function searchTicketTrack() {
            const query = document.getElementById('trackSearchInput').value.trim();
            renderTrackedTickets(query);
        }

        // Render Admin Table & Kanban Stats
        function renderAdminTable() {
            const tbody = document.getElementById('adminTableBody');
            tbody.innerHTML = '';

            const searchQuery = document.getElementById('adminSearchInput').value.toLowerCase();
            const catFilter = document.getElementById('adminCategoryFilter').value;
            const urgFilter = document.getElementById('adminUrgencyFilter').value;

            let filtered = state.tickets.filter(t => {
                const matchesSearch = t.refId.toLowerCase().includes(searchQuery) ||
                                      t.studentName.toLowerCase().includes(searchQuery) ||
                                      t.location.toLowerCase().includes(searchQuery);
                const matchesCat = catFilter === 'All' || t.category === catFilter;
                const matchesUrg = urgFilter === 'All' || t.urgency === urgFilter;
                return matchesSearch && matchesCat && matchesUrg;
            });

            // Update Header Stats
            document.getElementById('statTotalTickets').innerText = state.tickets.length;
            document.getElementById('statCriticalTickets').innerText = state.tickets.filter(t => t.urgency === 'Critical').length;
            document.getElementById('statPendingTickets').innerText = state.tickets.filter(t => t.status !== 'Resolved').length;
            document.getElementById('statResolvedTickets').innerText = state.tickets.filter(t => t.status === 'Resolved').length;
            document.getElementById('ticketCountBadge').innerText = `${filtered.length} items`;

            filtered.forEach(ticket => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-50/50 dark:hover:bg-slate-700/30 transition';

                const urgencyBadge = ticket.urgency === 'Critical'
                    ? 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-300'
                    : (ticket.urgency === 'Medium' ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300' : 'bg-emerald-100 text-emerald-700');

                tr.innerHTML = `
                    <td class="py-3 px-4 font-mono font-bold text-brand-600">${ticket.refId}</td>
                    <td class="py-3 px-4">
                        <div class="font-semibold text-slate-800 dark:text-slate-200">${ticket.studentName}</div>
                        <div class="text-[10px] text-slate-400">${ticket.department} (${ticket.studentId})</div>
                    </td>
                    <td class="py-3 px-4">
                        <div class="text-slate-700 dark:text-slate-300">${ticket.category}</div>
                        <div class="text-[10px] text-amber-600 font-medium">${ticket.location}</div>
                    </td>
                    <td class="py-3 px-4">
                        <span class="px-2 py-0.5 rounded text-[10px] font-bold ${urgencyBadge}">${ticket.urgency}</span>
                        <div class="text-[10px] text-slate-400 mt-0.5">SLA: ${ticket.slaHours}</div>
                    </td>
                    <td class="py-3 px-4">
                        <select onchange="updateTicketStatus('${ticket.refId}', this.value)" class="text-xs bg-slate-100 dark:bg-slate-900 border border-slate-200 dark:border-slate-700 rounded px-2 py-1 outline-none">
                            <option value="Pending" ${ticket.status === 'Pending' ? 'selected' : ''}>Pending</option>
                            <option value="Investigating" ${ticket.status === 'Investigating' ? 'selected' : ''}>Investigating</option>
                            <option value="In Progress" ${ticket.status === 'In Progress' ? 'selected' : ''}>In Progress</option>
                            <option value="Resolved" ${ticket.status === 'Resolved' ? 'selected' : ''}>Resolved</option>
                        </select>
                    </td>
                    <td class="py-3 px-4 text-right space-x-1">
                        <button onclick="previewEmailForTicket('${ticket.refId}')" class="p-1.5 hover:bg-brand-50 hover:text-brand-600 text-slate-500 rounded-lg transition" title="Preview Escalation Email">
                            <i data-lucide="mail" class="w-4 h-4"></i>
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
            lucide.createIcons();
        }

        function updateTicketStatus(refId, newStatus) {
            const ticket = state.tickets.find(t => t.refId === refId);
            if (ticket) {
                ticket.status = newStatus;
                showToast(`Status updated to ${newStatus}`);
                renderAdminTable();
                renderTrackedTickets();
            }
        }

        // Email Escalation Preview Modal Operations
        function previewEmailForTicket(refId) {
            const ticket = state.tickets.find(t => t.refId === refId);
            if (ticket) {
                state.currentActiveTicket = ticket;
                openEmailPreviewModal();
            }
        }

        function openEmailPreviewModal() {
            if (!state.currentActiveTicket) return;
            const t = state.currentActiveTicket;
            document.getElementById('modalEmailRecipient').innerText = t.emailDraft.to;
            document.getElementById('modalEmailSubject').innerText = t.emailDraft.subject;
            document.getElementById('modalEmailBody').innerText = t.emailDraft.body;
            document.getElementById('emailModal').classList.remove('hidden');
        }

        function closeEmailModal() {
            document.getElementById('emailModal').classList.add('hidden');
        }

        function copyModalEmail() {
            const body = document.getElementById('modalEmailBody').innerText;
            navigator.clipboard.writeText(body);
            showToast("Email text copied to clipboard!");
        }

        function downloadEmailTxt() {
            if (!state.currentActiveTicket) return;
            const t = state.currentActiveTicket;
            const content = `TO: ${t.emailDraft.to}\nSUBJECT: ${t.emailDraft.subject}\n\n${t.emailDraft.body}`;
            const blob = new Blob([content], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `Escalation_${t.refId}.txt`;
            a.click();
            showToast("Escalation email downloaded");
        }

        function simulateSendEmail() {
            const btn = document.getElementById('btnSimulateSend');
            btn.innerHTML = `<i data-lucide="loader" class="w-3.5 h-3.5 animate-spin"></i> Dispatching...`;
            setTimeout(() => {
                btn.innerHTML = `<i data-lucide="check" class="w-3.5 h-3.5"></i> Dispatched!`;
                showToast(`Escalation email dispatched to ${state.currentActiveTicket.emailDraft.to}`);
                setTimeout(() => {
                    btn.innerHTML = `<i data-lucide="send" class="w-3.5 h-3.5"></i> Dispatch Email`;
                    closeEmailModal();
                }, 1000);
            }, 1000);
        }

        // Export Data
        function exportData(type) {
            let dataStr = "";
            let filename = `Grievance_Report_${new Date().toISOString().slice(0,10)}`;

            if (type === 'json') {
                dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(state.tickets, null, 2));
                filename += ".json";
            } else {
                const headers = ["RefID", "StudentName", "StudentID", "Department", "Category", "Urgency", "Location", "Status", "Timestamp"];
                const rows = state.tickets.map(t => [t.refId, t.studentName, t.studentId, t.department, t.category, t.urgency, t.location, t.status, t.timestamp]);
                const csvContent = [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
                dataStr = "data:text/csv;charset=utf-8," + encodeURIComponent(csvContent);
                filename += ".csv";
            }

            const dlAnchor = document.createElement('a');
            dlAnchor.setAttribute("href", dataStr);
            dlAnchor.setAttribute("download", filename);
            document.body.appendChild(dlAnchor);
            dlAnchor.click();
            dlAnchor.remove();
            showToast(`Exported report as ${type.toUpperCase()}`);
        }

        // Render Chart.js Analytics
        function updateAnalytics() {
            const categories = {};
            const urgencies = { Critical: 0, Medium: 0, Low: 0 };
            const locationCounts = {};

            state.tickets.forEach(t => {
                categories[t.category] = (categories[t.category] || 0) + 1;
                urgencies[t.urgency] = (urgencies[t.urgency] || 0) + 1;
                locationCounts[t.location] = (locationCounts[t.location] || 0) + 1;
            });

            // 1. Category Chart
            const ctxCat = document.getElementById('categoryChart').getContext('2d');
            if (categoryChartInstance) categoryChartInstance.destroy();
            categoryChartInstance = new Chart(ctxCat, {
                type: 'doughnut',
                data: {
                    labels: Object.keys(categories),
                    datasets: [{
                        data: Object.values(categories),
                        backgroundColor: ['#2563eb', '#f59e0b', '#10b981', '#8b5cf6', '#ec4899']
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });

            // 2. Urgency Chart
            const ctxUrg = document.getElementById('urgencyChart').getContext('2d');
            if (urgencyChartInstance) urgencyChartInstance.destroy();
            urgencyChartInstance = new Chart(ctxUrg, {
                type: 'bar',
                data: {
                    labels: Object.keys(urgencies),
                    datasets: [{
                        label: 'Tickets',
                        data: Object.values(urgencies),
                        backgroundColor: ['#ef4444', '#f59e0b', '#10b981']
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });

            // 3. Hotspots List
            const hotspotsContainer = document.getElementById('hotspotsList');
            hotspotsContainer.innerHTML = '';
            Object.entries(locationCounts).forEach(([loc, count]) => {
                const div = document.createElement('div');
                div.className = 'p-3 rounded-xl bg-slate-50 dark:bg-slate-900 border border-slate-200 dark:border-slate-700 flex justify-between items-center text-xs';
                div.innerHTML = `
                    <span class="font-medium text-slate-800 dark:text-slate-200">${loc}</span>
                    <span class="px-2 py-0.5 bg-red-100 text-red-700 font-bold rounded-full text-[10px]">${count} issue(s)</span>
                `;
                hotspotsContainer.appendChild(div);
            });
        }
